#include<stdio.h>
#include<stdlib.h>
 main()
{
   int n,*ptr,i,sum=0;
   printf("Enter the number of element");
   scanf("%d",&n);
   ptr=(int*)malloc(n*sizeof(int));
   printf("Enter the element");
   for(i=0;i<n;i++)
   {
      scanf("%d",ptr+i);
      sum+=*(ptr+i);
   }
   printf("sum=%d",sum);
   free(ptr);
}
