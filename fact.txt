#include<stdio.h>
int factorial(int n)
{
   if(n==0 || n==1)
   {
      return 1;
   }
   else{
      return n*factorial(n-1);
   }
}
int main()
{
   int n,i;
   int fact =1;
   printf("Entr  the number:");
   scanf("%d",&n);
   fact=factorial(n);
   printf("factorial : %d",fact);
}
