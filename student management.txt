#include<stdio.h>
struct stu
{
   int roll;
   char n[20];
   int m1,m2,tot;
   float avg;
}x[100],y[100];
int main()
{
   FILE *fp;
   fp=fopen("x1.txt","w");
   int n,i;
   printf("Enter the no.of students:");
   scanf("%d",&n);
   for(i=0;i<n;i++)
   {
      printf("Enter the roll number:");
      scanf("%d",&x[i].roll);
      printf("Enter the name:");
      scanf("%s",&x[i].n);
      printf("Enter mark1:");
      scanf("%d",&x[i].m1);
      printf("Enter mark2:");
      scanf("%d",&x[i].m2);
      x[i].tot=x[i].m1+x[i].m2;
      x[i].avg=x[i].tot/2;
   }
   for(i=0;i<n;i++)
   {
     fprintf(fp,"%d\t %s\t %d\t %d\t %d\t %f\n",x[i].roll,x[i].n,x[i].m1,x[i].m2,x[i].tot,x[i].avg);
   }
   fclose(fp);
   fp=fopen("x1.txt","r");
   for(i=0;i<n;i++)
   {
     fscanf(fp,"%d\t %s\t %d\t %d\t %d\t %f\n",&y[i].roll,&y[i].n,&y[i].m1,&y[i].m2,&y[i].tot,&y[i].avg);
     if(y[i].avg>90)
     {
      printf("%d\t %s\t %d\t %d\t %d\t %f\n",y[i].roll,y[i].n,y[i].m1,y[i].m2,y[i].tot,y[i].avg);
     }
   }
fclose(fp);
}
