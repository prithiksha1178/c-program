include<stdio.h>
#include<stdlib.h>
int main()
{
   char std[50],ad1[50],ad2[50];
   int n,i,*p;
   printf("Enter the no of students:\n");
   scanf("%d",&n);
   for(i=0;i<n;i++)
   {
      std[i]=i+1;
      printf("\n The roll no of %d student:%d",i+1,std[i]);
   }
   for(i=0;i<n;i++)
   {
       *p=std[i];
        if((i+1)%2==0)
        {
            ad1[i]=*p;
            printf("\n the first  staff advisor's student with roll no:%d",ad1[i]);
            p++;
         }
         else
         {
            ad2[i]=*p;
            printf("\n the secound staff advisor's student with roll no:%d",ad2[i]);
         }
   }
   return 0;
}
